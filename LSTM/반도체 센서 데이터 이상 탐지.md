# 1. 데이터 로드 및 전처리
```
import pandas as pd
import numpy as np
import torch
import torch.nn as nn
from sklearn.preprocessing import StandardScaler
from torch.utils.data import TensorDataset, DataLoader

url = "https://archive.ics.uci.edu/ml/machine-learning-databases/secom/secom.data"
label_url = "https://archive.ics.uci.edu/ml/machine-learning-databases/secom/secom_labels.data"

data = pd.read_csv(url, sep=" ", header=None)
labels = pd.read_csv(label_url, sep=" ", header=None)

data = data.fillna(0)
y = labels[0].replace(-1, 0).values

scaler = StandardScaler()
data_scaled = scaler.fit_transform(data)

print(data_scaled.shape)
# (1567, 590)
```
* 스케일링이란
  * 서로 다른 단위와 범위를 가진 데이터들을 일정한 규칙에 따라 맞추는 작업
  * 압력 센서(1000~2000 pa), 온도 센서(20~30℃) -> 이 상태로 LSTM에 넣으면, 모델은 숫자가 큰 압력 센서가 더 중요한 정보라고 오해하게 되고 온도의 미세한 변화는 압력의 변화에 묻혀버리게 된다. 이를 방지하기 위해 모든 데이터를 0~1 사이나 평균 0, 표준편차 1의 동일한 범위에서 뛰게 만드는 것.
* 주요 스케일링 기법
  * StandardScaler (표준화): 평균을 0, 분산을 1로 만든다. 데이터가 종 모양(가우시안 분포)일 때 효과적이며, 이상치에 조금 더 강하다.(현재 코드에서 사용중인)
  * MinMaxScaler (정규화): 모든 값을 0과 1 사이로 압축. 데이터의 최소/최대값이 명확할 때 좋다.

# 2. 시계열 윈도우
```
def create_sequences(data, labels, seq_length):
  X_seq, y_seq = [], []
  for i in range(len(data) - seq_length):
    X_seq.append(data[i:i+seq_length])  # 과거 seq_length 만큼의 센서 기록
    y_seq.append(labels[i+seq_length])  # 바로 다음 시점의 불량 여부
  return np.array(X_seq), np.array(y_seq)

SEQ_LENGTH = 10
X_samples, y_samples = create_sequences(data_scaled, y, SEQ_LENGTH)

X_tensor = torch.tensor(X_samples, dtype=torch.float32)
y_tensor = torch.tensor(y_samples, dtype=torch.float32).unsqueeze(1)

dataset = TensorDataset(X_tensor, y_tensor)
train_loader = DataLoader(dataset, batch_size=32, shuffle=True)

print(f"LSTM 입력 형태: {X_tensor.shape}")
# LSTM 입력 형태: torch.Size([1557, 10, 590])
```
* 데이터를 시계열 형태로 조립하는 과정
* X_seq.append(data[i:i+seq_length])
  * 현재 시점($i$)부터 $seq\_length$만큼의 데이터를 잘라서 저장
  * 예를 들어 seq_length가 10이라면, 0~9번까지의 센서 데이터를 하나의 학습 단위로 묶는 것.
* y_seq.append(labels[i + seq_length])
  * 0~9번 데이터를 보고 예측해야 할 정답($y$)은 바로 다음 시점인 10번의 상태(정상/불량)이다. 그래서 인덱스에 seq_length를 더해준 값을 가져온다.
* 시퀸스 길이를 설정하는 이유
  * 반도체 공정에서 갑자기 불량이 발생하는 경우는 드물다. 보통 온도가 서서히 올라가거나, 압력이 조금씩 흔들리는 등의 전조 현상이 먼저 나타난다.
  * 시퀸스 길이를 1로 하면 지금 이 순간의 센서값(일반 머신러닝)만 보지만, 길이를 10으로 하면 최근 10번의 흐름을 보고 다음 상태를 파악할 수 있다.
  * 즉, 과거의 데이터 흐름을 통해 미래를 예측하기 위해 사용한다.

# 3. LSTM 모델 학습 및 평가
```
class AnomalyDetector(nn.Module):
  def __init__(self, input_dim, hidden_dim, num_layers):
    super(AnomalyDetector, self).__init__()

    # 1. 하이퍼 파라미터 저장
    self.hidden_dim = hidden_dim
    self.num_layers = num_layers

    # 2. LSTM 레이어: 시계열 데이터의 흐름을 파악하는 중심
    # batch_first=True는 입력 데이터가 (배치, 시퀸스, 특징) 순서임을 알려줌
    self.lstm = nn.LSTM(input_dim, hidden_dim, num_layers, batch_first=True)

    # 3. Fully Connected Layer: LSTM의 마지막 기억을 받아 불량 확률(0~1)로 변환
    self.fc = nn.Linear(hidden_dim, 1)

  def forward(self, x):
    # x의 모양: (batch_size, seq_length, 590)

    # h0, c0: 초기 은닉 상태와 셀 상태를 0으로 설정
    h0 = torch.zeros(self.num_layers, x.size(0), self.hidden_size).to(x.device)
    c0 = torch.zeros(self.num_layers, x.size(0), self.hidden_size).to(x.device)

    # LSTM 통과: out은 모든 시점의 출력값들을 담고 있음
    out, (hn, cn) = self.lstm(x)

    # 마지막 시점의 흐름 요약만 필요
    # out[:, -1, :] -> 모든 배치에 대해, 마지막(-1) 시퀸스의 hidden state만 추출
    final_output = self.fc(out[:, -1, :])

    return final_output
```
* 입력: 590개의 센서가 10초(시퀸스 길이)동안 들어온다.(반복)
* LSTM 내부: 첫 번째 1초의 데이터를 보고 Cell_state(cn)에 담는다. 그다음 2초 데이터가 들어오면 기존 정보와 합쳐서 업데이트. 이 과정을 10초까지 반복
* 최종 판단: 10초까지 다본 후의 마지막 상태값(out[:, -1, :])만 꺼낸다.
* 출력: : Linear 레이어가 이 보고서를 읽고 최종 점수 출력

```
device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
model = AnomalyDetector(input_dim=590, hidden_dim=64, num_layers=2).to(device)

# 불균형 데이터 처리를 위해 이상(1) 라벨에 더 높은 가중치 부여
# 정상 대비 이상 데이터 비율이 낮기 때문.
pos_weight = torch.tensor([10.0]).to(device)
criterion = nn.BCEWithLogitsLoss(pos_weight=pos_weight) # Sigmoid가 내장된 손실함수
optimizer = torch.optim.Adam(model.parameters(), lr=0.001)

# 2. 학습 루프
num_epochs = 50
history = {'train_loss': []}

model.train()
for epoch in range(num_epochs):
  epoch_loss = 0
  for batch_X, batch_y in train_loader:
    batch_X, batch_y = batch_X.to(device), batch_y.to(device)

    outputs = model(batch_X)
    loss = criterion(outputs, batch_y)

    optimizer.zero_grad()
    loss.backward()
    optimizer.step()

    epoch_loss += loss.item()

  avg_loss = epoch_loss / len(train_loader)
  history['train_loss'].append(avg_loss)

  if (epoch+1) % 10 == 0:
    print(f'Epoch [{epoch+1}/{num_epochs}], Loss: {avg_loss:.4f}')

plt.plot(history['train_loss'])
plt.title('Model Loss Progress')
plt.xlabel('Epoch')
plt.ylabel('Loss')
plt.show()
```

# 4. 모델 평가 및 Confusion Matrix 시각화
```
from sklearn.metrics import confusion_matrix, classification_report
import seaborn as sns

# 1. 평가 모드로 전환 및 예측값 수집
model.eval()
all_preds = []
all_labels = []

with torch.no_grad(): # 평가 시에는 기울기 계산을 하지 않음
  for batch_X, batch_y in train_loader:
    batch_X = batch_X.to(device)
    outputs = model(batch_X)

    # Logit을 확률(0~1)로 변환 후 0.5 기준으로 0 또는 1 결정
    preds = (torch.sigmoid(outputs) > 0.5).float()

    all_preds.extend(preds.cpu().numpy())
    all_labels.extend(batch_y.numpy())

# 2. Confusion Matrix 생성
cm = confusion_matrix(all_labels, all_preds)

# 3. 시각화
plt.figure(figsize=(8, 6))
sns.heatmap(cm, annot=True, fmt='d', cmap='Blues',
            xticklabels=['Normal(0)', 'Anomaly(1)'],
            yticklabels=['Normal(0)', 'Anomaly(1)'])
plt.xlabel('Predicted')
plt.ylabel('Actual')
plt.title('Confusion Matrix: Semiconductor Anomaly Detection')
plt.show()

print(classification_report(all_labels, all_preds, target_names=['Normal(0)', 'Anomaly(1)']))
```
